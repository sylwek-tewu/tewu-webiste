/**
 * Rich text of the FAQ answers: `[label](href)` is a link and `**text**` is bold, nothing else.
 * One parser feeds the page (RichText), the JSON-LD answers and the content tests.
 */
export type RichSegment =
  | { kind: 'text'; text: string }
  | { kind: 'strong'; text: string }
  | { kind: 'link'; text: string; href: string };

const TOKEN = /\[([^\]]+)\]\(([^)\s]+)\)|\*\*(.+?)\*\*/g;

export function parseRichText(text: string): RichSegment[] {
  const segments: RichSegment[] = [];
  let last = 0;
  for (const match of text.matchAll(TOKEN)) {
    if (match.index > last) segments.push({ kind: 'text', text: text.slice(last, match.index) });
    if (match[1] !== undefined) segments.push({ kind: 'link', text: match[1], href: match[2] });
    else segments.push({ kind: 'strong', text: match[3] });
    last = match.index + match[0].length;
  }
  if (last < text.length) segments.push({ kind: 'text', text: text.slice(last) });
  return segments;
}

export const isExternalHref = (href: string) => /^https?:\/\//.test(href);

const escapeHtml = (text: string) =>
  text.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');

/** HTML for JSON-LD answers: site paths become absolute, so the answer stands on its own. */
export function richTextToHtml(text: string, origin: string): string {
  return parseRichText(text)
    .map((segment) => {
      if (segment.kind === 'text') return escapeHtml(segment.text);
      if (segment.kind === 'strong') return `<strong>${escapeHtml(segment.text)}</strong>`;
      const href = isExternalHref(segment.href) ? segment.href : `${origin}${segment.href}`;
      return `<a href="${escapeHtml(href)}">${escapeHtml(segment.text)}</a>`;
    })
    .join('');
}
```

import type { Locale } from '@/i18n/types';

export interface PluralForms {
  one: string;
  few: string;
  many: string;
}

/**
 * Polish and Ukrainian plural form of a count. They differ only for 21, 31, …:
 * „21 pytań” but „21 запитання”.
 */
export function pluralForm(locale: Locale, n: number): keyof PluralForms {
  const lastDigit = n % 10;
  const lastTwo = n % 100;
  const few = lastDigit >= 2 && lastDigit <= 4 && (lastTwo < 12 || lastTwo > 14);
  const one = locale === 'pl' ? n === 1 : lastDigit === 1 && lastTwo !== 11;
  if (one) return 'one';
  return few ? 'few' : 'many';
}

export function formatCount(locale: Locale, n: number, forms: PluralForms): string {
  return `${n} ${forms[pluralForm(locale, n)]}`;
}
```

import type { Locale } from '@/i18n/types';

const MONTHS_GENITIVE: Record<Locale, string[]> = {
  pl: ['stycznia', 'lutego', 'marca', 'kwietnia', 'maja', 'czerwca', 'lipca', 'sierpnia', 'września', 'października', 'listopada', 'grudnia'],
  uk: ['січня', 'лютого', 'березня', 'квітня', 'травня', 'червня', 'липня', 'серпня', 'вересня', 'жовтня', 'листопада', 'грудня'],
};

const YEAR_ABBREVIATION: Record<Locale, string> = { pl: 'r.', uk: 'р.' };

/**
 * "2026-10-15" → "15 października 2026 r." / "15 жовтня 2026 р.". Without Intl, so the server
 * and every browser render the same text (no hydration mismatch).
 */
export function formatLongDate(locale: Locale, isoDate: string): string {
  const [year, month, day] = isoDate.split('-').map(Number);
  return `${day} ${MONTHS_GENITIVE[locale][month - 1]} ${year} ${YEAR_ABBREVIATION[locale]}`;
}

// Global Constraints: no "tax advice" wording, in any inflection and word order.
const FORBIDDEN_PHRASES = [/doradztw\p{L}*\s+podatkow/iu, /podatkow\p{L}*\s+doradztw/iu, /податков\p{L}*\s+консульт/iu, /консульт\p{L}*\s+податков/iu];

function allText(content: FaqPageContent): string[] {
  const texts: string[] = [];
  const walk = (value: unknown) => {
    if (typeof value === 'string') texts.push(value);
    else if (Array.isArray(value)) value.forEach(walk);
    else if (value && typeof value === 'object') Object.values(value).forEach(walk);
  };
  walk(content);
  return texts;
}

          </Anchor>

      expect(fs.existsSync(path.join(PUBLIC, file)), file).toBe(true);
    }
  });

  it.each(Object.entries(OFFICE_PHOTOS))('%s: the JPEG has the registered size and no EXIF block', (_name, photo) => {
    const jpeg = fs.readFileSync(path.join(PUBLIC, photo.jpg));
    // SOF0/SOF2 marker: height and width are the 2-byte values after the precision byte
    let offset = 2;
    let size: { width: number; height: number } | null = null;
    let hasExif = false;
    while (offset < jpeg.length && !size) {
      const marker = jpeg.readUInt16BE(offset);
      const length = jpeg.readUInt16BE(offset + 2);
      if (marker === 0xffe1) hasExif = true;
      if (marker === 0xffc0 || marker === 0xffc2) size = { height: jpeg.readUInt16BE(offset + 5), width: jpeg.readUInt16BE(offset + 7) };
      offset += 2 + length;
    }
    expect(size).toEqual({ width: photo.width, height: photo.height });
    expect(hasExif).toBe(false);
  });
});

function useOpenQuestionFromHash() {
  useEffect(() => {
    const openFromHash = () => {
      const id = decodeURIComponent(window.location.hash.slice(1));
      const target = id ? document.getElementById(id) : null;
      if (!(target instanceof HTMLDetailsElement)) return;
      target.open = true;
      target.scrollIntoView?.({ block: 'start' });
    };
    openFromHash();
    window.addEventListener('hashchange', openFromHash);
    return () => window.removeEventListener('hashchange', openFromHash);
  }, []);
}

function faqAnswerHtml({ answer, sources }: FaqQuestion, sourcesLabel: FaqPageContent['sourcesLabel']): string {
  const paragraphs = answer.map((paragraph) => `<p>${richTextToHtml(paragraph, SITE_URL)}</p>`);
  if (sources?.length) {
    const label = sources.length === 1 ? sourcesLabel.one : sourcesLabel.many;
    const links = sources.map(({ label: text, href }) => richTextToHtml(`[${text}](${href})`, SITE_URL)).join(', ');
    paragraphs.push(`<p>${label}: ${links}</p>`);
  }
  return paragraphs.join('');
}

/**
 * The FAQ page and its breadcrumbs. FAQPage is itself a WebPage, so one node carries the page's
 * address, language and date as well as the questions.
 */
export function buildFaqPageJsonLd(content: FaqPageContent, updatedAt: string) {
  const { locale } = content;
  const url = absolute(localizePath('/faq', locale));
  const questions = content.categories.flatMap((category) => category.questions);
  return {
    '@context': 'https://schema.org',
    '@graph': [
      {
        '@type': 'FAQPage',
        url,
        name: content.hero.title,
        description: content.meta.description,
        inLanguage: IN_LANGUAGE[locale],
        dateModified: updatedAt,
        isPartOf: { '@id': WEBSITE_ID },
        about: { '@id': COMPANY_ID },
        mainEntity: questions.map((question) => ({
          '@type': 'Question',
          name: question.question,
          url: `${url}#${question.id}`,
          acceptedAnswer: { '@type': 'Answer', text: faqAnswerHtml(question, content.sourcesLabel) },
        })),
      },
      {
        '@type': 'BreadcrumbList',
        itemListElement: [
          { '@type': 'ListItem', position: 1, name: BREADCRUMB_LABELS[locale].home, item: homeUrl(locale) },
          { '@type': 'ListItem', position: 2, name: content.breadcrumb, item: url },
        ],
      },
    ],
  };
}

/** Throws for a page without an entry above, so a new page cannot slip out of llms.txt. */
function pageLine(path: string): string {
  const slug = SERVICE_PAGE_SLUGS.find((s) => path === `/uslugi/${s}`);
  if (slug) return `- [${SERVICE_PAGES[slug].label}](${url(path)}): usługa biura rachunkowego w Szczecinie`;
  const page = PAGES[path];
  if (!page) throw new Error(`llms.txt: no description for ${path}`);
  return `- [${page.name}](${url(path)}): ${page.description}`;
}

    '',
    `Kontakt: tel. ${phones.office[0].display}, e-mail ${emails.main}, pn–pt ${openingHours.opens}–${openingHours.closes}.`,
    '',
    '## Strony',
    '',
    ...main.map(pageLine),
    '',
    '## Wersja ukraińska',
    '',
    `- [Головна](${url('/uk')}): бухгалтерське бюро в Щецині, обслуговування українською`,
    `- [Поширені запитання (FAQ)](${url('/uk/faq')}): JDG чи Sp. z o.o., форма оподаткування, VAT, ZUS, рахунок, KSeF`,
    '',
    '## Optional',
    '',
    pageLine('/polityka-prywatnosci'),
    '',
  ].join('\n');
}
